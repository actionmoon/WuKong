# 分布式网络对象管理器

> **说明**：本文档仅提供 Shared Mode 下分布式动态网络对象管理器的代码骨架与最佳实践。关于分布式与中心化的理论对比及选型原则，请参阅 Manual 手册中的《网络对象的动态创建》。

## 1. 核心管理器组件 (NetworkObjectManager.lua)

全局网络对象管理器，负责动态孵化、注册与销毁，需挂载至场景全局节点。

```Lua
-- NetworkObjectManager.lua
local networkObjects = {} -- 网络对象字典，以 uid 为 key

function Awake()
    -- 确保单例初始化
    if _G["NetworkObjectManager"] ~= nil then
        if self.gameObject ~= nil then UnityEngine.Object.Destroy(self.gameObject) end
        return
    end
    _G["NetworkObjectManager"] = self.script
end

-- 动态创建网络对象并绑定 ActorID
function Spawn(objectKey, bindActorID, action, position, rotation)
    local onSpawn = function(go, id)
        local networkScript = go:GetDouyinScript("NetworkObject")
        if bindActorID ~= nil and networkScript ~= nil then
            -- 绑定生命周期
            networkScript.script.syncActorID:SetValue(bindActorID)
        end
        action?(go, id) -- 执行创建回调
    end
    
    -- 根据参数调用不同 NetSpawn 重载
    if position and rotation then
        return CS.DouyinObjectService.NetSpawn(objectKey, position, rotation, onSpawn)
    elseif position then
        return CS.DouyinObjectService.NetSpawn(objectKey, position, onSpawn)
    else
        return CS.DouyinObjectService.NetSpawn(objectKey, onSpawn)
    end
end

-- 注册网络对象（由网络对象自身在 OnNetSpawned 中调用）
function OnNetworkObjectSpawned(networkObj)
    networkObjects[networkObj.uid] = networkObj
end

-- 反注册网络对象（由网络对象自身在 OnDestroy 中调用）
function OnNetworkObjectDestroy(uid)
    networkObjects[uid] = nil
end

-- 主权方主动销毁网络对象
function Despawn(uid)
    local object = networkObjects[uid]
    if object == nil then return end
    
    if object:HasOwnership() then
        CS.DouyinObjectService.NetDestroy(object.gameObject)
    else
        -- 无主权时，通知主权方代为销毁（需处理丢包异常）
        self:SendMessageToTarget("DespawnImp", object:GetOwner(), uid)
    end
end

-- 接收无主权方请求，直接销毁网络对象
function DespawnImp(uid)
    local object = networkObjects[uid]
    if object == nil then return end
    CS.DouyinObjectService.NetDestroy(object.gameObject, true)
end

-- 对象查找接口
function GetNetworkObject(uid) return networkObjects[uid] end
function GetNetworkObjects() return networkObjects end
```

## 2. 网络对象基类 (NetworkObject.lua)

挂载于需要被管理的动态预制体上（如血条、特效等）。

```Lua
-- NetworkObject.lua
---@varSync syncActorID : int = 0

function OnNetSpawned()
    _G.NetworkObjectManager.OnNetworkObjectSpawned(self)
end

-- 1. 最后一个用户正常退出时，清理因房间留存导致的网络对象
-- 2. 网络对象所在端异常退出时，其他主权用户代为销毁
function Update()
    if not self:HasOwnership() then return end
    
    local actorID = syncActorID:GetValue()
    if actorID == 0 then return end
    
    -- 若绑定的 Actor 已离线，自动销毁残留对象
    local actor = DouyinActorService.GetActorById(actorID)
    if actor == nil then
        CS.DouyinObjectService.NetDestroy(self.gameObject)
    end
end

function OnDestroy()
    if _G.NetworkObjectManager then
        _G.NetworkObjectManager.OnNetworkObjectDestroy(self.uid)
    end
end
```

### 3.2 血量视图脚本 (BloodText.lua)

挂载至血条预制体，与 `NetworkObject.lua` 同节点。

```Lua
-- BloodText.lua
local bloodText = nil
local offsetY = 2.0

function OnNetSpawned()
    -- 从同节点的 NetworkObject 获取绑定的目标 ActorID
    local targetActorId = self:GetDouyinScript("NetworkObject").script.syncActorID:GetValue()
    InitBlood(targetActorId)
end

function InitBlood(targetActorIdValue)
    local targetActor = DouyinActorService.GetActorById(targetActorIdValue)
    if not targetActor then return end
    
    _G.PlayerBloodMgr.RegisterBloodView(targetActorIdValue, self)
    
    -- 设为玩家子对象并更新局部偏移
    self.transform:SetParent(targetActor.gameObject.transform)
    self.transform.localPosition = Vector3(0, offsetY, 0)
    
    bloodText = self.transform:Find("text"):GetComponent(CS.TMPro.TextMeshPro)
    bloodText.text = tostring(bloodValue:GetValue())
end

-- 监听血量同步变量变化，实时更新文本 UI
bloodValue:OnValueChange(function(curVal, preVal)
    if bloodText then
        bloodText.text = tostring(curVal)
    end
end)
```
